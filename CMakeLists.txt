cmake_minimum_required(VERSION 3.10)
project(lord_sumi_embedded_core LANGUAGES CXX)
enable_testing()
add_executable(embedded_core_test cpp/tests/test_core.cpp)
target_include_directories(embedded_core_test PRIVATE cpp/include)
set_target_properties(embedded_core_test PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(embedded_core_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME embedded_core COMMAND embedded_core_test)
